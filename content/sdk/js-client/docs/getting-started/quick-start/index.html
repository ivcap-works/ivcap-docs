    Quick Start¶

    This guide walks you through your first complete IVCAP workflow in TypeScript: discover a service, upload data, run a job, and download the results.

    1. Install the SDK¶

    npm install @ivcap/client

    2. Create a Client¶

    import { IvcapClient } from "@ivcap/client";

const client = new IvcapClient({
  baseUrl: "https://develop.ivcap.net",
  token: process.env.IVCAP_JWT,
});

    Or rely on environment variables (IVCAP_URL and IVCAP_JWT):

    const client = new IvcapClient(); // reads env vars automatically

    3. Browse Available Services¶

    // Iterate all services (lazy pagination)
for await (const svc of client.services.list()) {
  console.log(svc.id, svc.name);
}

// Fetch details about a specific service
const svc = await client.services.get("urn:ivcap:service:abc-123");
console.log(svc.description);
svc.parameters.forEach(p => {
  console.log(`  ${p.name} (${p.type})${p.optional ? " [optional]" : ""}`);
});

    4. Upload an Input Artifact¶

    const artifact = await client.artifacts.upload(
  JSON.stringify({ temperature: [21.3, 22.1, 19.8] }),
  { name: "sensor-data.json", contentType: "application/json" }
);
console.log("Uploaded artifact:", artifact.id);

    You can also upload a file in Node.js:

    import { createReadStream } from "fs";

const artifact = await client.artifacts.upload(
  createReadStream("data.csv"),
  { name: "data.csv", contentType: "text/csv" }
);

    5. Submit a Job and Wait for Results¶

    const job = await client.jobs.createAndWait(
  "urn:ivcap:service:my-analysis-service",
  {
    parameters: [
      { name: "input",     value: artifact.id },
      { name: "threshold", value: "0.05" },
    ],
    name: "My first job",
  },
  {
    pollInterval: 5_000,   // check every 5 seconds
    timeout:     300_000,  // give up after 5 minutes
    onUpdate: (j) => console.log("status:", j.status),
  }
);

console.log("Final status:", job.status);

    6. Download the Results¶

    if (job.status === "succeeded") {
  for (const product of job.products.items) {
    console.log("Product:", product.id, product.mimeType);

    // Download as text
    const text = await client.artifacts.downloadText(product.id);
    console.log(text);

    // Or download as parsed JSON
    const obj = await client.artifacts.downloadJson(product.id);
    console.log(obj);
  }
} else {
  console.error("Job did not succeed:", job.status);
}

    Complete Example¶

    import { IvcapClient, JobResource } from "@ivcap/client";

async function main() {
  const client = new IvcapClient({
    baseUrl: "https://develop.ivcap.net",
    token: process.env.IVCAP_JWT,
  });

  // 1. Find the service
  const serviceId = "urn:ivcap:service:my-analysis-service";
  const svc = await client.services.get(serviceId);
  console.log("Using service:", svc.name);

  // 2. Upload input data
  const artifact = await client.artifacts.upload(
    JSON.stringify({ values: [1, 2, 3, 4, 5] }),
    { name: "input.json", contentType: "application/json" }
  );

  // 3. Submit & wait
  const job = await client.jobs.createAndWait(
    serviceId,
    {
      parameters: JobResource.buildParameters({
        input:     artifact.id,
        threshold: "0.5",
      }),
      name: "Quick-start run",
    },
    { onUpdate: (j) => console.log("  →", j.status) }
  );

  // 4. Download results
  if (job.status === "succeeded") {
    for (const p of job.products.items) {
      const result = await client.artifacts.downloadJson(p.id);
      console.log("Result:", result);
    }
  }
}

main().catch(console.error);
